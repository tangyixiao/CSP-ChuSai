const int N = 2e5 + 5;
int n, a[N], tmp[N];
void merge_sort(int l, int r) {
    if (l >= r)
        return;
    int mid = (l + r) >> 1;
    merge_sort(l, mid);
    merge_sort(mid + 1, r);
    int i = l, j = mid + 1, k = l;
    while (i <= mid && j <= r) {
        if (a[i] <= a[j])
            tmp[k++] = a[i++];
        else
            tmp[k++] = a[j++];
    }
    while (i <= mid)
        tmp[k++] = a[i++];
    while (j <= r)
        tmp[k++] = a[j++];
    for (int p = l; p <= r; ++p)
        a[p] = tmp[p];
}
int main() {

#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, head[N], to[N * 2], nxt[N * 2], ec;
int dista[N], que[N];
void add(int u, int v) {
    to[++ec] = v;
    nxt[ec] = head[u];
    head[u] = ec;
}
int bfs(int start) {
    fill(dista + 1, dista + n + 1, -1);
    int l = 0, r = 0, far = start;
    que[r++] = start;
    dista[start] = 0;
    while (l < r) {
        int u = que[l++];
        if (dista[u] > dista[far])
            far = u;
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (dista[v] == -1) {
                dista[v] = dista[u] + 1;
                que[r++] = v;
            }
        }
    }
    return far;
}
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n;
    for (int i = 1, u, v; i < n; ++i) {
        cin >> u >> v;
        add(u, v);
        add(v, u);
    }
    int s = bfs(1);
    int t = bfs(s);
    cout << dista[t];
}
\end{lstlisting}\end{codebox}

\noindent（1）（有序序列的第 K 小和）有两个长度均为 \texttt{n} 的单调不降序列 \texttt{a} 和 \texttt{b}。从两个序列中各取一个数相加，共得到 \texttt{n\string^2} 个和，求这些和中第 \texttt{k} 小的值。输入满足 \texttt{1 <= n <= 10\string^5}，\texttt{1 <= k <= n\string^2}，元素为非负整数且答案不超过 \texttt{10\string^9}。
\begin{codebox}\begin{lstlisting}
#include <bits/stdc++.h>
using namespace std;
const int N = 100005;
int n, a[N], b[N];
long long k;
int upper_bound_index(int *a, int n, long long x) {
    int l = 0, r = |@\blank{①}@|;
    while (l < r) {
        int mid = (l + r) >> 1;
        if (|@\blank{②}@|)
            r = mid;
        else
            l = mid + 1;
    }
    return |@\blank{③}@|;
}
long long count_le(long long sum) {
    long long cnt = 0;
    for (int i = 0; i < n; ++i)
        cnt += upper_bound_index(b, n, sum - a[i]);
    return cnt;
}
long long solve() {
    long long l = |@\blank{④}@|, r = |@\blank{⑤}@|;
    while (l < r) {
        long long mid = (l + r) >> 1;
        if (count_le(mid) >= k)
            r = mid;
        else
            l = mid + 1;
    }
    return l;
}
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> k;
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < n; ++i) cin >> b[i];
    cout << solve();
}
\end{lstlisting}\end{codebox}

\optw{A. \texttt{a[mid] > x}}{B. \texttt{a[mid] >= x}}\optw{C. \texttt{a[mid] < x}}{D. \texttt{a[mid] <= x}}
\q{36.} ③处应填（\qquad）\\
\optw{A. \texttt{l}}{B. \texttt{l+1}}\optw{C. \texttt{r-1}}{D. \texttt{n-l}}
\q{37.} ④处应填（\qquad）\\
\optw{A. \texttt{a[0]+b[0]}}{B. \texttt{a[1]+b[1]}}\optw{C. \texttt{0}}{D. \texttt{a[n]+b[n]}}
\q{38.} ⑤处应填（\qquad）\\
\optw{A. \texttt{a[n-1]+b[n-1]}}{B. \texttt{a[n]+b[n]}}\optw{C. \texttt{2*N}}{D. \texttt{n-1}}

\noindent（2）（最小生成树）给定一个有 \texttt{n} 个结点、\texttt{m} 条带权无向边的图，求其最小生成树的边权和；若图不连通则输出 \texttt{Impossible}。输入满足 \texttt{1 <= n <= 2 * 10\string^5}，\texttt{1 <= m <= 4 * 10\string^5}。
\begin{codebox}\begin{lstlisting}
#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5, M = 4e5 + 5;
struct Edge {
    int u, v, w;
    bool operator < (const Edge &other) const {
        return w < other.w;
    }
} e[M];
int n, m, fa[N];
int find(int x) {
    return fa[x] == x ? x : fa[x] = find(fa[x]);
}
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> m;
    for (int i = 1; i <= n; ++i)
        fa[i] = |@\blank{①}@|;
    for (int i = 0; i < m; ++i)
        cin >> e[i].u >> e[i].v >> e[i].w;
    sort(e, e + |@\blank{②}@|);
    long long ans = 0;
    int used = 0;
    for (int i = 0; i < m; ++i) {
        int x = find(e[i].u), y = find(e[i].v);
        if (|@\blank{③}@|)
            continue;
        fa[x] = |@\blank{④}@|;
        ans += e[i].w;
        ++used;
    }
    if (used != |@\blank{⑤}@|)
        cout << "Impossible";
    else
        cout << ans;
}
\end{lstlisting}\end{codebox}

\q{40.} ②处应填（\qquad）\\

\q{41.} ③处应填（\qquad）\\
\optw{A. \texttt{x == y}}{B. \texttt{x != y}}\optw{C. \texttt{x < y}}{D. \texttt{e[i].u == e[i].v}}
\q{42.} ④处应填（\qquad）\\
\optw{A. \texttt{x}}{B. \texttt{y}}\optw{C. \texttt{e[i].v}}{D. \texttt{fa[y]}}
\q{43.} ⑤处应填（\qquad）\\
\optw{A. \texttt{n}}{B. \texttt{n-1}}\optw{C. \texttt{m}}{D. \texttt{used+1}}
