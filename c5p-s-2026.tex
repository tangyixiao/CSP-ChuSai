#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, a[N], tmp[N];
long long ans;
void merge_sort(int l, int r) {
    if (l >= r)
        return;
    int mid = (l + r) >> 1;
    merge_sort(l, mid);
    merge_sort(mid + 1, r);
    int i = l, j = mid + 1, k = l;
    while (i <= mid && j <= r) {
        if (a[i] <= a[j])
            tmp[k++] = a[i++];
        else {
            tmp[k++] = a[j++];
            ans += mid - i + 1;
        }
    }
    while (i <= mid)
        tmp[k++] = a[i++];
    while (j <= r)
        tmp[k++] = a[j++];
    for (int p = l; p <= r; ++p)
        a[p] = tmp[p];
}
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n;
    for (int i = 1; i <= n; ++i)
        cin >> a[i];
    merge_sort(1, n);
    cout << ans;
}

\q{27.} 虽然程序中存在嵌套的 \texttt{while}，但每个字符最多入栈一次、出栈一次，因此程序的总时间复杂度为 \texttt{O(n)}。（\qquad）\\

\begin{codebox}\begin{lstlisting}
#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, head[N], to[N * 2], nxt[N * 2], ec;
int dista[N], que[N];
void add(int u, int v) {
    to[++ec] = v;
    nxt[ec] = head[u];
    head[u] = ec;
}
int bfs(int start) {
    fill(dista + 1, dista + n + 1, -1);
    int l = 0, r = 0, far = start;
    que[r++] = start;
    dista[start] = 0;
    while (l < r) {
        int u = que[l++];
        if (dista[u] > dista[far])
            far = u;
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (dista[v] == -1) {
                dista[v] = dista[u] + 1;
                que[r++] = v;
            }
        }
    }
    return far;
}
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n;
    for (int i = 1, u, v; i < n; ++i) {
        cin >> u >> v;
        add(u, v);
        add(v, u);
    }
    int s = bfs(1);
    int t = bfs(s);

}

\noindent（1）（有序序列的第 K 小和）有两个长度均为 \texttt{n} 的单调不降序列 \texttt{a} 和 \texttt{b}。从两个序列中各取一个数相加，共得到 \texttt{n\string^2} 个和，求这些和中第 \texttt{k} 小的值。输入满足 \texttt{1 <= n <= 10\string^5}，\texttt{1 <= k <= n\string^2}，元素为非负整数且答案不超过 \texttt{10\string^9}。程序用双指针在线性时间内计算“不超过某个值的数对数量”。
\begin{codebox}\begin{lstlisting}
#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int n, a[N], b[N];
long long k;
long long count_le(long long x) {
    long long cnt = 0;
    int j = |@\blank{①}@|;
    for (int i = 1; i <= n; ++i) {
        while (|@\blank{②}@|)
            --j;
        cnt += |@\blank{③}@|;
    }
    return cnt;
}
long long solve() {
    long long l = |@\blank{④}@|;
    long long r = 1LL * a[n] + b[n];
    while (l + 1 < r) {
        long long mid = (l + r) >> 1;
        if (count_le(mid) >= k)
            r = mid;
        else
            l = mid;
    }
    return |@\blank{⑤}@|;
}
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> k;
    for (int i = 1; i <= n; ++i)
        cin >> a[i];
    for (int i = 1; i <= n; ++i)
        cin >> b[i];
    cout << solve();
}
\end{lstlisting}\end{codebox}

\noindent（2）（减法游戏与 SG 函数）有 \texttt{n} 堆石子和 \texttt{m} 种允许的取石数量 \texttt{d[1..m]}。每次操作选择一堆，并从中恰好取走某个 \texttt{d[j]} 颗石子（该堆石子数必须不少于 \texttt{d[j]}）；无法操作的人判负。各堆互不影响，双方均采用最优策略。输入满足 \texttt{1 <= n <= 10\string^5}，\texttt{1 <= m <= 20}，\texttt{0 <= a[i] <= 10\string^5}。下面程序计算每个石子数的 SG 函数，并判断先手是否必胜。
\begin{codebox}\begin{lstlisting}
#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int n, m, d[25], a[N], sg[N];
int vis[25], tim;
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> m;
    for (int i = 1; i <= m; ++i)
        cin >> d[i];
    int mx = 0;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        mx = max(mx, a[i]);
    }
    for (int x = 1; x <= mx; ++x) {
        ++tim;
        for (int j = 1; j <= m; ++j) {
            if (|@\blank{①}@|)
                vis[|@\blank{②}@|] = tim;
        }
        int g = 0;
        while (|@\blank{③}@|)
            ++g;
        sg[x] = |@\blank{④}@|;
    }
    int xr = 0;
    for (int i = 1; i <= n; ++i)
        xr ^= |@\blank{⑤}@|;
    cout << (xr ? "Win" : "Lose");
}
\end{lstlisting}\end{codebox}
