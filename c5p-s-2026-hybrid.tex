using namespace std;
const int P = 998244353, M = 10;
int n, m, ban[105], f[2][1 << M];
vector<int> st;

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) {
        string s; cin >> s;
        for (int j = 0; j < m; ++j)
            if (s[j] == '#') ban[i] |= 1 << j;
    }
    for (int s = 0; s < (1 << m); ++s)
        if ((s & (s << 1)) == 0) st.push_back(s);
    f[0][0] = 1;
    for (int i = 1; i <= n; ++i) {
        int cur = i & 1, pre = cur ^ 1;
        memset(f[cur], 0, sizeof(f[cur]));
        for (int s : st) {
            if (s & ban[i]) continue;
            for (int t : st) {
                if (s & t) continue;
                f[cur][s] += f[pre][t];
                if (f[cur][s] >= P) f[cur][s] -= P;
            }
        }
    }
    int ans = 0;
    for (int s : st) {
        ans += f[n & 1][s];
        if (ans >= P) ans -= P;
    }
    cout << ans << '\n';
}

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll P = 1000000007;
const int N = 2e5 + 5;
int n, q;
ll a[N];
struct Node { ll sum, mul, add; } tr[N << 2];

void build(int p, int l, int r) {
    tr[p].mul = 1;
    if (l == r) { tr[p].sum = a[l] % P; return; }
    int m = (l + r) >> 1;
    build(p << 1, l, m); build(p << 1 | 1, m + 1, r);
    tr[p].sum = (tr[p << 1].sum + tr[p << 1 | 1].sum) % P;
}
void apply(int p, int l, int r, ll k, ll b) {
    tr[p].sum = (tr[p].sum * k + (r - l + 1) * b) % P;
    tr[p].mul = tr[p].mul * k % P;
    tr[p].add = (tr[p].add * k + b) % P;
}
void push(int p, int l, int r) {
    if (l == r) return;
    int m = (l + r) >> 1;
    apply(p << 1, l, m, tr[p].mul, tr[p].add);
    apply(p << 1 | 1, m + 1, r, tr[p].mul, tr[p].add);
    tr[p].mul = 1; tr[p].add = 0;
}
void upd(int p, int l, int r, int ql, int qr, ll k, ll b) {
    if (ql <= l && r <= qr) { apply(p, l, r, k, b); return; }
    push(p, l, r);
    int m = (l + r) >> 1;
    if (ql <= m) upd(p << 1, l, m, ql, qr, k, b);
    if (qr > m) upd(p << 1 | 1, m + 1, r, ql, qr, k, b);
    tr[p].sum = (tr[p << 1].sum + tr[p << 1 | 1].sum) % P;
}
ll qry(int p, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return tr[p].sum;
    push(p, l, r);
    int m = (l + r) >> 1;
    ll s = 0;
    if (ql <= m) s = (s + qry(p << 1, l, m, ql, qr)) % P;
    if (qr > m) s = (s + qry(p << 1 | 1, m + 1, r, ql, qr)) % P;
    return s;
}
int main() {
    cin >> n >> q;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    build(1, 1, n);
    while (q--) {
        int op, l, r; cin >> op >> l >> r;
        if (op == 1) {
            ll k, b; cin >> k >> b;
            upd(1, 1, n, l, r, k, b);
        } else cout << qry(1, 1, n, l, r) << '\n';
    }
}

\noindent（1）（有序序列的第 K 小和）有两个长度均为 \texttt{n} 的单调不降序列 \texttt{a} 和 \texttt{b}。从两个序列中各取一个数相加，共得到 \texttt{n\string^2} 个和，求这些和中第 \texttt{k} 小的值。输入满足 \texttt{1 <= n <= 10\string^5}，\texttt{1 <= k <= n\string^2}，元素为非负整数且答案不超过 \texttt{10\string^9}。
\begin{codebox}\begin{lstlisting}
#include <bits/stdc++.h>
using namespace std;
const int N = 100005;
int n, a[N], b[N];
long long k;
int upper_bound_index(int *a, int n, long long x) {
    int l = 0, r = |@\blank{①}@|;
    while (l < r) {
        int mid = (l + r) >> 1;
        if (|@\blank{②}@|)
            r = mid;
        else
            l = mid + 1;
    }
    return |@\blank{③}@|;
}
long long count_le(long long sum) {
    long long cnt = 0;
    for (int i = 0; i < n; ++i)
        cnt += upper_bound_index(b, n, sum - a[i]);
    return cnt;
}
long long solve() {
    long long l = |@\blank{④}@|, r = |@\blank{⑤}@|;
    while (l < r) {
        long long mid = (l + r) >> 1;
        if (count_le(mid) >= k)
            r = mid;
        else
            l = mid + 1;
    }
    return l;
}
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> k;
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < n; ++i) cin >> b[i];
    cout << solve();
}
\end{lstlisting}\end{codebox}

\noindent（2）（最小生成树）给定一个有 \texttt{n} 个结点、\texttt{m} 条带权无向边的图，求其最小生成树的边权和；若图不连通则输出 \texttt{Impossible}。输入满足 \texttt{1 <= n <= 2 * 10\string^5}，\texttt{1 <= m <= 4 * 10\string^5}。
\begin{codebox}\begin{lstlisting}
#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5, M = 4e5 + 5;
struct Edge {
    int u, v, w;
    bool operator < (const Edge &other) const {
        return w < other.w;
    }
} e[M];
int n, m, fa[N];
int find(int x) {
    return fa[x] == x ? x : fa[x] = find(fa[x]);
}
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> m;
    for (int i = 1; i <= n; ++i)
        fa[i] = |@\blank{①}@|;
    for (int i = 0; i < m; ++i)
        cin >> e[i].u >> e[i].v >> e[i].w;
    sort(e, e + |@\blank{②}@|);
    long long ans = 0;
    int used = 0;
    for (int i = 0; i < m; ++i) {
        int x = find(e[i].u), y = find(e[i].v);
        if (|@\blank{③}@|)
            continue;
        fa[x] = |@\blank{④}@|;
        ans += e[i].w;
        ++used;
    }
    if (used != |@\blank{⑤}@|)
        cout << "Impossible";
    else
        cout << ans;
}
\end{lstlisting}\end{codebox}
